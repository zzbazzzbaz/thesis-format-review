```markdown
# 规则资产目录

本目录是格式审核系统的正式规则资产入口。

## 目录结构

- `*/rule.yml`：运行时规则元数据，只包含规则编号、标题、严重程度、状态、说明、批注文案和期望值。
- `*/case.yml`：测试样本清单，只用于测试和规则资产校验，不参与生产审核。

- `rule.yml` 不包含 `object`、`category`、`scope` 字段。
- `case.yml` 必须写明 `description` 和 `error_description`。
- 已发布的规则编号不能改名、不能复用；不用的规则标记为 `deprecated`。
```

```python
"""规则元数据模型。"""
from __future__ import annotations

import re
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, field_validator

RuleSeverity = Literal["error", "warning", "info"]
RuleStatus = Literal["draft", "active", "deprecated"]
_RULE_ID_RE = re.compile(r"^[a-z]+(?:-[a-z]+)*-\d{3}$")


class RuleSpec(BaseModel):
    model_config = ConfigDict(extra="forbid")

    id: str
    title: str
    severity: RuleSeverity
    status: RuleStatus
    description: str
    message: str
    expected: str

    @field_validator("id")
    @classmethod
    def validate_rule_id(cls, value: str) -> str:
        if not _RULE_ID_RE.fullmatch(value):
            raise ValueError("rule id must be lowercase words joined by '-' and end with three digits")
        return value


class RuleModuleSpec(BaseModel):
    model_config = ConfigDict(extra="forbid")

    module: str
    module_name: str
    rules: list[RuleSpec] = Field(min_length=1)


class RuleCatalog(BaseModel):
```

```python
"""加载运行时 rule.yml 规则元数据。"""
from __future__ import annotations

from pathlib import Path
from typing import Any

import yaml
from pydantic import ValidationError

from thesis_review.rules.catalog.models import RuleCatalog, RuleModuleSpec, RuleSpec


class RuleCatalogError(ValueError):
    """规则目录配置错误。"""


def _read_yaml(path: Path) -> dict[str, Any]:
    try:
        data = yaml.safe_load(path.read_text(encoding="utf-8"))
    except yaml.YAMLError as exc:
        raise RuleCatalogError(f"Invalid YAML: {path}") from exc
    if not isinstance(data, dict):
        raise RuleCatalogError(f"YAML root must be a mapping: {path}")
    return data


def _load_module(path: Path) -> RuleModuleSpec:
    try:
        module = RuleModuleSpec.model_validate(_read_yaml(path))
    except ValidationError as exc:
        raise RuleCatalogError(f"Invalid rule.yml: {path}") from exc
    expected_module = path.parent.name
    if module.module != expected_module:
        raise RuleCatalogError(
            f"{path}: module must match directory name "
            f"({module.module!r} != {expected_module!r})"
        )
    return module


def load_rule_catalog(root: str | Path) -> RuleCatalog:
    root_path = Path(root)
    rule_files = sorted(root_path.glob("*/rule.yml"))
    if not rule_files:
        raise RuleCatalogError(f"No rule.yml files found under {root_path}")

    rules_by_id: dict[str, RuleSpec] = {}
    modules_by_name: dict[str, RuleModuleSpec] = {}
    for path in rule_files:
        module = _load_module(path)
        modules_by_name[module.module] = module
        for rule in module.rules:
            if rule.id in rules_by_id:
                raise RuleCatalogError(f"Duplicate rule id: {rule.id}")
            rules_by_id[rule.id] = rule

    return RuleCatalog(
        rules_by_id=rules_by_id,
        modules_by_name=modules_by_name,
    )
```

```python
def load_rule_cases(root: str | Path, catalog: RuleCatalog) -> list[RuleCaseSpec]:
    root_path = Path(root)
    case_files = sorted(root_path.glob("*/case.yml"))
    cases: list[RuleCaseSpec] = []

        try:
            case_file = RuleCaseFile.model_validate(_read_yaml(path))

                    )
```

```python
from __future__ import annotations

from thesis_review.models import Issue, Location
from thesis_review.rules.catalog import RuleCatalog


class UnknownRuleIdError(KeyError):
    """规则代码引用了不存在的 rule id。"""


class IssueFactory:
    def __init__(self, catalog: RuleCatalog):
        self.catalog = catalog

    def create(
        self,
        rule_id: str,
        actual: str | None,
        location: Location,
    ) -> Issue:
        try:
            rule = self.catalog.get(rule_id)
        except KeyError as exc:
            raise UnknownRuleIdError(rule_id) from exc
        return Issue(
            rule_id=rule.id,
            category="format",
            severity=rule.severity,
            message=rule.message,
            expected=rule.expected,
            actual=actual,
            location=location,
        )
```

```python
        "table-of-contents-title-006",
    )

    def check(self, context: RuleContext) -> list[Issue]:
        title = _toc_title(context.document)
        if title is None:
            return [
                context.issue_factory.create(
                    rule_id="table-of-contents-title-001",
                    actual="未找到目录",
                    location=_document_start_location(context.document),
                )
            ]

        issues: list[Issue] = []
        if title.text != "目    录":
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-title-001",
                    actual=title.text,
                    location=_block_location(title),
                )
            )
        if title.alignment != "center":
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-title-004",
                    actual=_display_alignment(title.alignment),
                    location=_block_location(title),
                )
            )
        if title.line_spacing is not None and not isclose(
            title.line_spacing,
            1.5,
            abs_tol=0.01,
        ):
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-title-005",
                    actual=_display_line_spacing(title.line_spacing),
                    location=_block_location(title),
                )
            )
        for run in title.runs:
            if not run.text.strip():
                continue
            if run.font_east_asian and run.font_east_asian != "黑体":
                issues.append(
                    context.issue_factory.create(
                        rule_id="table-of-contents-title-003",
                        actual=run.font_east_asian,
                        location=_block_location(title),
                    )
                )
            if run.size_pt is not None and not isclose(
                run.size_pt,
                16.0,
                abs_tol=0.01,
            ):
                issues.append(
                    context.issue_factory.create(
                        rule_id="table-of-contents-title-002",
                        actual=_display_size(run.size_pt),
                        location=_block_location(title),
                    )
                )
        blank_count = _empty_count_after(context.document, title)
        if blank_count != 1:
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-title-006",
                    actual=f"{blank_count} 个空段落",
                    location=_block_location(title),
                )
            )
        return issues


class TableOfContentsEntryChecker:
    checker_id = "table-of-contents-entry"
    rule_ids = (
        "table-of-contents-entry-001",
        "table-of-contents-entry-002",
        "table-of-contents-entry-003",
        "table-of-contents-entry-004",
        "table-of-contents-entry-005",
        "table-of-contents-entry-006",
        "table-of-contents-entry-007",
        "table-of-contents-entry-008",
    )

    def check(self, context: RuleContext) -> list[Issue]:
        if "table-of-contents" not in context.sections:
            return []
        issues: list[Issue] = []
        for para in _entry_paragraphs(context):
            self._check_spacing(context, para, issues)
            self._check_paragraph_format(context, para, issues)
            for run in para.runs:
                self._check_run(context, para, run, issues)
        return issues

    def _check_spacing(
        self,
        context: RuleContext,
        para: ParagraphBlock,
        issues: list[Issue],
    ) -> None:
        match = _FIRST_LEVEL_RE.match(para.text)
        if match is not None:
            spaces = match.group(2)
            if len(spaces) != 2:
                issues.append(
                    context.issue_factory.create(
                        rule_id="table-of-contents-entry-001",
                        actual=f"{len(spaces)} 个空格",
                        location=_block_location(para),
                    )
                )
            return

        match = _LOWER_LEVEL_RE.match(para.text)
        if match is None:
            return
        spaces = match.group(2)
        level = match.group(1).count(".") + 1
        rule_id = (
            "table-of-contents-entry-002"
            if level == 2
            else "table-of-contents-entry-003"
        )
        if len(spaces) != 1:
            issues.append(
                context.issue_factory.create(
                    rule_id=rule_id,
                    actual=f"{len(spaces)} 个空格",
                    location=_block_location(para),
                )
            )

    def _check_paragraph_format(
        self,
        context: RuleContext,
        para: ParagraphBlock,
        issues: list[Issue],
    ) -> None:
        if para.line_spacing is not None and not isclose(
            para.line_spacing,
            1.5,
            abs_tol=0.01,
        ):
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-entry-007",
                    actual=_display_line_spacing(para.line_spacing),
                    location=_block_location(para),
                )
            )

    def _check_run(
        self,
        context: RuleContext,
        para: ParagraphBlock,
        run: RunSpan,
        issues: list[Issue],
    ) -> None:
        if not run.text.strip() or run.text == "\t":
            return
        location = _run_location(para, run)
        if run.font_east_asian and run.font_east_asian != "宋体":
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-entry-004",
                    actual=run.font_east_asian,
                    location=location,
                )
            )
        if run.font_latin and run.font_latin != "Times New Roman":
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-entry-005",
                    actual=run.font_latin,
                    location=location,
                )
            )
        if run.size_pt is not None and not isclose(
            run.size_pt,
            12.0,
            abs_tol=0.01,
        ):
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-entry-006",
                    actual=_display_size(run.size_pt),
                    location=location,
                )
            )
        if run.bold:
            issues.append(
                context.issue_factory.create(
                    rule_id="table-of-contents-entry-008",
                    actual="加粗",
                    location=location,
                )
            )
```
