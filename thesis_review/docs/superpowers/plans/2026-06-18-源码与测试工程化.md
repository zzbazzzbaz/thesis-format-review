```python
"""审核结果模型。"""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import TypedDict

from thesis_review.core.models import Issue


class IssueSummary(TypedDict):
    total: int
    error: int
    warning: int
    info: int


@dataclass(frozen=True)
class ReviewResult:
    input_docx: Path
    out_dir: Path
    marked_docx: Path
    issues_json: Path
    issues: list[Issue]

    @property
    def summary(self) -> IssueSummary:
        return summarize_issues(self.issues)


def summarize_issues(issues: list[Issue]) -> IssueSummary:
    summary: IssueSummary = {
        "total": len(issues),
        "error": 0,
        "warning": 0,
        "info": 0,
    }
    for issue in issues:
        summary[issue.severity] += 1
    return summary
```

```python
"""审核结果 JSON 报告。"""
from __future__ import annotations

import json
from typing import Any

from thesis_review.engine.result import ReviewResult


def result_payload(result: ReviewResult) -> dict[str, Any]:
    return {
        "input": str(result.input_docx),
        "marked_docx": str(result.marked_docx),
        "summary": result.summary,
        "issues": [
            issue.model_dump(mode="json")
            for issue in result.issues
        ],
    }


def write_issues_json(result: ReviewResult) -> None:
    result.issues_json.parent.mkdir(parents=True, exist_ok=True)
    result.issues_json.write_text(
        json.dumps(result_payload(result), ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
```

```python
"""审核编排入口。"""
from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Protocol

from thesis_review.core.models import DocumentModel, Issue
from thesis_review.engine.report import write_issues_json
from thesis_review.engine.result import ReviewResult


class SessionFactory(Protocol):
    def __call__(self) -> Any:
        ...


@dataclass(frozen=True)
class ReviewDependencies:
    session_factory: SessionFactory
    extract: Callable[[Any, str], DocumentModel]
    annotate: Callable[[Any, str, str, list[Issue]], None]
    build_rule_context: Callable[[DocumentModel], Any]
    get_checkers: Callable[[], list[Any]]


def default_dependencies() -> ReviewDependencies:
    from thesis_review.annotator import annotate
    from thesis_review.extractor import extract
    from thesis_review.rules.context import build_rule_context
    from thesis_review.rules.registry import get_checkers
    from thesis_review.uno_session import UnoSession

    return ReviewDependencies(
        session_factory=UnoSession,
        extract=extract,
        annotate=annotate,
        build_rule_context=build_rule_context,
        get_checkers=get_checkers,
    )


def review(
    input_docx: str | Path,
    out_dir: str | Path,
    *,
    dependencies: ReviewDependencies | None = None,
) -> ReviewResult:
    input_path = Path(input_docx)
    output_dir = Path(out_dir)
    marked_docx = output_dir / "marked.docx"
    issues_json = output_dir / "issues.json"
    output_dir.mkdir(parents=True, exist_ok=True)

    deps = dependencies or default_dependencies()
    with deps.session_factory() as session:
        with session.open(str(input_path)) as doc:
            document = deps.extract(doc, source_path=str(input_path))
        context = deps.build_rule_context(document)
        issues: list[Issue] = []
        for checker in deps.get_checkers():
            issues.extend(checker.check(context))
        deps.annotate(session, str(input_path), str(marked_docx), issues)

    result = ReviewResult(
        input_docx=input_path,
        out_dir=output_dir,
        marked_docx=marked_docx,
        issues_json=issues_json,
        issues=issues,
    )
    write_issues_json(result)
    return result
```

```python
"""命令行入口。"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path

from thesis_review.engine import review


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="thesis-review",
        description="毕业设计论文格式审核",
    )
    subparsers = parser.add_subparsers(dest="command", required=True)
    review_parser = subparsers.add_parser("review", help="审核 .docx 文档")
    review_parser.add_argument("input", help="输入 .docx 路径")
    review_parser.add_argument("--out-dir", required=True, help="输出目录")
    return parser


def validate_input(path: str | Path) -> Path:
    file_path = Path(path)
    if not file_path.exists():
        print(f"错误：文件不存在：{file_path}", file=sys.stderr)
        raise SystemExit(2)
    if file_path.suffix.lower() != ".docx":
        print("错误：仅支持 .docx 输入(不支持 .doc，不做转换)", file=sys.stderr)
        raise SystemExit(2)
    return file_path


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    if args.command == "review":
        input_path = validate_input(args.input)
        out_dir = Path(args.out_dir)
        result = review(input_path, out_dir)
```

```python
"""上传文件保存、审核调用和临时文件生命周期管理。"""
from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path
import shutil
import tempfile

from fastapi import UploadFile

from thesis_review.engine import review

CHUNK_SIZE = 1024 * 1024


class InvalidUpload(ValueError):
    """用户上传内容不符合首版 Web 接口要求。"""


@dataclass(frozen=True)
class WebReviewResult:
    output_path: Path
    issues_json: Path
    download_name: str
    cleanup: Callable[[], None]


def _client_filename(filename: str | None) -> str:
    if not filename:
        raise InvalidUpload("请选择 .docx 文件")
    name = filename.replace("\\", "/").split("/")[-1].strip()
    if not name:
        raise InvalidUpload("请选择 .docx 文件")
    if Path(name).suffix.lower() != ".docx":
        raise InvalidUpload("仅支持 .docx 文件")
    return name


def _download_name(filename: str) -> str:
    stem = Path(filename).stem
    return f"{stem}.marked.docx"


async def _save_upload(upload: UploadFile, target: Path) -> None:
    with target.open("wb") as f:
        while True:
            chunk = await upload.read(CHUNK_SIZE)
            if not chunk:
                break
            f.write(chunk)


async def review_upload(upload: UploadFile) -> WebReviewResult:
    filename = _client_filename(upload.filename)
    workdir = Path(tempfile.mkdtemp(prefix="thesis-review-"))
    input_path = workdir / "input.docx"
    out_dir = workdir / "review"

    def cleanup() -> None:
        shutil.rmtree(workdir, ignore_errors=True)

    try:
        await _save_upload(upload, input_path)
        result = review(str(input_path), out_dir)
    except Exception:
        cleanup()
        raise

    return WebReviewResult(
        output_path=result.marked_docx,
        issues_json=result.issues_json,
        download_name=_download_name(filename),
        cleanup=cleanup,
    )
```

```python
"""规则 case.yml 测试辅助。"""
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import yaml

from thesis_review.rules.catalog import RuleCaseFile, RuleCaseSpec, RuleCatalog


@dataclass(frozen=True)
class ResolvedRuleCase:
    module: str
    case: RuleCaseSpec
    module_dir: Path
    sample_path: Path
    active_rule_ids: set[str]


def resolve_rule_cases(rules_root: Path, catalog: RuleCatalog) -> list[ResolvedRuleCase]:
    resolved: list[ResolvedRuleCase] = []
    for module_dir in sorted(path for path in rules_root.iterdir() if path.is_dir()):
        case_path = module_dir / "case.yml"
        if not case_path.exists():
            continue
        case_file = RuleCaseFile.model_validate(
            yaml.safe_load(case_path.read_text(encoding="utf-8"))
        )
        module_spec = catalog.modules_by_name[module_dir.name]
        active_rule_ids = {
            rule.id for rule in module_spec.rules
            if rule.status == "active"
        }
        for case in case_file.cases:
            for rule_id in case.expected_rule_ids:
                if rule_id not in catalog.rules_by_id:
                    raise AssertionError(f"Unknown rule id in {case_path}: {rule_id}")
            resolved.append(
                ResolvedRuleCase(
                    module=module_dir.name,
                    case=case,
                    module_dir=module_dir,
                    sample_path=module_dir / case.file,
                    active_rule_ids=active_rule_ids,
                )
            )
    return resolved


def filter_module_rule_ids(rule_ids: set[str], active_rule_ids: set[str]) -> set[str]:
    return rule_ids & active_rule_ids
```
