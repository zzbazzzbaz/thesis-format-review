# 毕业设计格式审核系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用 LibreOffice/UNO 引擎实现一个生产级毕业设计论文格式审核系统：读 `.docx` → 跑格式/结构规则 → 把问题以红色底纹+批注回写为 `marked.docx`。

**Architecture:** 三步解耦管道——`extractor`(UNO 读)产出引擎无关的 `DocumentModel`；`rules`(纯 Python 注册表)只读 `DocumentModel` 产出 `Issue`；`annotator`(UNO 写)只读 `Issue` 写回文档。读写同在一个 LibreOffice 引擎内，定位契约单引擎闭环。

**Tech Stack:** Python 3.11+、pydantic、LibreOffice + UNO(pyuno，非 pip 包)、pytest(强制 TDD)、python-docx(仅 dev，用于程序化生成测试 fixture)。

**实现顺序原则：** 先做纯 Python、可快速 TDD 的层(models → rules)，把价值和测试覆盖立起来；再做依赖 LibreOffice 环境的 UNO 层(uno_session → extractor → annotator)；最后管道、CLI、部署。规则层不依赖 UNO，因此能在没有任何 UNO 代码时就完整测通。

**前置环境：** 本机/CI 需安装 LibreOffice。本机 soffice 路径：`/Applications/LibreOffice.app/Contents/MacOS/soffice`(mac)。UNO 测试用 pytest marker `@pytest.mark.uno` 标记，便于在无 LibreOffice 环境下跳过。

**代码详细度约定(分层)：**
- **纯 Python 层(Task 1-10：models / rules / registry / cli)**：步骤中的代码是**确定实现**，可直接采用。
- **UNO 层(Task 12-17：uno_session / extractor / annotator / roundtrip / pipeline)**：步骤中的代码是**参考实现**，不是标准答案。UNO 的确切 API（属性名如 `CharFontNameAsian`、行距 `Mode` 枚举值、批注服务名等）**以实际跑通为准**——实现时若与真实 UNO 行为不符，按真实行为修改代码，**验收标准始终是该任务的测试断言**(例如 extractor 必须从 sample.docx 读出 `楷体`/`14pt`，roundtrip 必须通过)，而非逐字符匹配参考代码。遇到 UNO API 不确定时，先写最小探针脚本验证属性名再落地。

```python
"""引擎无关的核心数据模型，是各层之间唯一的耦合点。"""
from __future__ import annotations

from pydantic import BaseModel


class RunSpan(BaseModel):
    id: str
    text: str
    run_index: int
    font_east_asian: str | None
    font_latin: str | None
    size_pt: float | None
    bold: bool
    italic: bool


class ParagraphBlock(BaseModel):
    id: str
    text: str
    paragraph_index: int
    style_name: str | None
    outline_level: int | None
    alignment: str | None
    line_spacing: float | None
    first_line_indent_chars: float | None
    runs: list[RunSpan]


class DocumentModel(BaseModel):
    source_docx: str
    paragraphs: list[ParagraphBlock]


class Location(BaseModel):
    kind: str
    block_id: str
    paragraph_index: int
    run_id: str | None = None
    run_index: int | None = None


class Issue(BaseModel):
    rule_id: str
    category: str
    severity: str
    message: str
    expected: str | None = None
    actual: str | None = None
    location: Location
```

```python
"""格式规则：字体、字号、行距、缩进。"""
from __future__ import annotations

from thesis_review.models import DocumentModel, Issue, Location


class BodyFontRule:
    rule_id = "format.body.font"
    category = "format"
    severity = "error"
    EXPECTED_FONT = "宋体"

    def check(self, document: DocumentModel) -> list[Issue]:
        issues: list[Issue] = []
        for para in document.paragraphs:
            if para.outline_level is not None:
                continue
            for run in para.runs:
                if run.font_east_asian and run.font_east_asian != self.EXPECTED_FONT:
                    issues.append(Issue(
                        rule_id=self.rule_id, category=self.category,
                        severity=self.severity,
                        message=f"正文中文字体应为{self.EXPECTED_FONT}",
                        expected=self.EXPECTED_FONT, actual=run.font_east_asian,
                        location=Location(
                            kind="run", block_id=para.id,
                            paragraph_index=para.paragraph_index,
                            run_id=run.id, run_index=run.run_index)))
        return issues
```

```python
class RequiredSectionRule:
```

```python
"""管理 headless soffice 进程与 UNO 连接：单实例串行。"""
from __future__ import annotations

import contextlib
import os
import subprocess
import time
from pathlib import Path

import uno
from com.sun.star.beans import PropertyValue

_DEFAULT_SOFFICE = [
    "/Applications/LibreOffice.app/Contents/MacOS/soffice",  # mac
    "/usr/bin/soffice",                                       # linux
    "soffice",                                                # PATH
]
_CONN = "socket,host=127.0.0.1,port=2002;urp;StarOffice.ComponentContext"


def _find_soffice() -> str:
    env = os.environ.get("SOFFICE_BIN")
    if env:
        return env
    for cand in _DEFAULT_SOFFICE:
        if cand == "soffice" or Path(cand).exists():
            return cand
    raise RuntimeError("找不到 soffice，可设 SOFFICE_BIN 指定路径")


def _to_url(path: str) -> str:
    return uno.systemPathToFileUrl(str(Path(path).resolve()))


class UnoSession:
    def __init__(self) -> None:
        self._proc: subprocess.Popen | None = None
        self._desktop = None

    def __enter__(self) -> "UnoSession":
        self._start()
        return self

    def __exit__(self, *exc) -> None:
        self.close()

    def _start(self) -> None:
        soffice = _find_soffice()
        self._proc = subprocess.Popen([
            soffice, "--headless", "--norestore", "--invisible",
            "--nodefault", "--nologo",
            f"--accept={_CONN}",
        ])
        ctx_local = uno.getComponentContext()
        resolver = ctx_local.ServiceManager.createInstanceWithContext(
            "com.sun.star.bridge.UnoUrlResolver", ctx_local)
        last_err: Exception | None = None
        for _ in range(60):  # 最多等 ~30s 让 soffice 起来
            try:
                ctx = resolver.resolve(
                    "uno:socket,host=127.0.0.1,port=2002;urp;"
                    "StarOffice.ComponentContext")
                smgr = ctx.ServiceManager
                self._desktop = smgr.createInstanceWithContext(
                    "com.sun.star.frame.Desktop", ctx)
                return
            except Exception as e:  # noqa: BLE001 - 连接前会持续失败
                last_err = e
                time.sleep(0.5)
        raise RuntimeError(f"无法连接 soffice UNO 桥：{last_err}")

    def is_ready(self) -> bool:
        return self._desktop is not None

    @contextlib.contextmanager
    def open(self, path: str):
        props = (PropertyValue(Name="Hidden", Value=True),)
        doc = self._desktop.loadComponentFromURL(
            _to_url(path), "_blank", 0, props)
        try:
            yield doc
        finally:
            with contextlib.suppress(Exception):
                doc.close(False)

    def close(self) -> None:
        if self._desktop is not None:
            with contextlib.suppress(Exception):
                self._desktop.terminate()
            self._desktop = None
        if self._proc is not None:
            with contextlib.suppress(Exception):
                self._proc.terminate()
                self._proc.wait(timeout=10)
            self._proc = None
```

```python
"""UNO 读 → DocumentModel。唯一读文档引擎的入口之一。"""
from __future__ import annotations

from thesis_review.models import RunSpan, ParagraphBlock, DocumentModel

_BOLD = 150.0  # com.sun.star.awt.FontWeight.BOLD
_ITALIC = 2    # com.sun.star.awt.FontSlant.ITALIC


def _is_paragraph(elem) -> bool:
    return elem.supportsService("com.sun.star.text.Paragraph")


def _extract_runs(para, block_id: str) -> list[RunSpan]:
    runs: list[RunSpan] = []
    enum = para.createEnumeration()
    idx = 0
    while enum.hasMoreElements():
        portion = enum.nextElement()
        if getattr(portion, "TextPortionType", "Text") != "Text":
            continue
        text = portion.getString()
        if text == "":
            continue
        runs.append(RunSpan(
            id=f"{block_id}-r-{idx:04d}", text=text, run_index=idx,
            font_east_asian=getattr(portion, "CharFontNameAsian", None) or None,
            font_latin=getattr(portion, "CharFontName", None) or None,
            size_pt=float(portion.CharHeight) if portion.CharHeight else None,
            bold=float(getattr(portion, "CharWeight", 100.0)) >= _BOLD,
            italic=int(getattr(portion, "CharPosture", 0)) == _ITALIC,
        ))
        idx += 1
    return runs


def extract(doc, source_path: str) -> DocumentModel:
    paragraphs: list[ParagraphBlock] = []
    enum = doc.Text.createEnumeration()
    p_idx = 0
    while enum.hasMoreElements():
        elem = enum.nextElement()
        if not _is_paragraph(elem):
            continue
        block_id = f"p-{p_idx:04d}"
        runs = _extract_runs(elem, block_id)
        paragraphs.append(ParagraphBlock(
            id=block_id, text=elem.getString(), paragraph_index=p_idx,
            style_name=getattr(elem, "ParaStyleName", None),
            outline_level=None,           # Task 14 补全
            alignment=None,               # Task 14 补全
            line_spacing=None,            # Task 14 补全
            first_line_indent_chars=None, # Task 14 补全
            runs=runs,
        ))
        p_idx += 1
    return DocumentModel(source_docx=source_path, paragraphs=paragraphs)
```

## Task 14: extractor — 大纲级别、对齐、行距、首行缩进换算

**说明：** 补全段落级属性。重点是单位换算：UNO 缩进单位是 1/100 mm，"字符数"需结合字号换算(1 字符 ≈ 字号磅值；1pt = 0.352778mm)。大纲级别来自 `para.OutlineLevel`(UNO：0=正文，1=一级标题…)，需转成模型约定(None=正文，0=一级…)。

**UNO 属性映射：**
- 大纲级别：`para.OutlineLevel`(0=正文/body，1=Heading1…)；模型：`outline_level = uno_level - 1 if uno_level >= 1 else None`
- 对齐：`para.ParaAdjust`(com.sun.star.style.ParagraphAdjust：LEFT=0,RIGHT=1,BLOCK=2,CENTER=3)
- 行距：`para.ParaLineSpacing`(结构体，`.Mode` 与 `.Height`；PROP 模式下 Height=150 表示 1.5 倍)
- 首行缩进：`para.ParaFirstLineIndent`(1/100 mm，整数)

```python
    return (lvl - 1) if lvl >= 1 else None

    return _ADJUST.get(int(getattr(para, "ParaAdjust", 0)), None)


def _line_spacing(para):
    ls = getattr(para, "ParaLineSpacing", None)
    if ls is None:
        return None
    if int(ls.Mode) == _LINE_PROP:
```

```python
"""UNO 写：红色底纹 + 批注 → marked.docx。唯一写文档的模块。"""
from __future__ import annotations

import logging

import uno
from com.sun.star.beans import PropertyValue

from thesis_review.models import Issue

log = logging.getLogger(__name__)
_RED = 0xFF0000


def _paragraphs(doc) -> list:
    out = []
    enum = doc.Text.createEnumeration()
    while enum.hasMoreElements():
        elem = enum.nextElement()
        if elem.supportsService("com.sun.star.text.Paragraph"):
            out.append(elem)
    return out


def _portions(para) -> list:
    out = []
    enum = para.createEnumeration()
    while enum.hasMoreElements():
        p = enum.nextElement()
        if getattr(p, "TextPortionType", "Text") == "Text" and p.getString():
            out.append(p)
    return out


def _target_range(doc, paras, issue: Issue):
    p_idx = issue.location.paragraph_index
    if p_idx >= len(paras):
        return None
    para = paras[p_idx]
    if issue.location.kind == "run" and issue.location.run_index is not None:
        portions = _portions(para)
        r_idx = issue.location.run_index
        if r_idx >= len(portions):
            return None
        return portions[r_idx]
    return para


def _add_comment(doc, target, message: str) -> None:
    ann = doc.createInstance("com.sun.star.text.textfield.Annotation")
    ann.Content = message
    ann.Author = "审核系统"
    cursor = doc.Text.createTextCursorByRange(target.getStart())
    doc.Text.insertTextContent(cursor, ann, False)


def annotate(session, input_docx: str, output_docx: str,
             issues: list[Issue]) -> None:
    with session.open(input_docx) as doc:
        paras = _paragraphs(doc)
        for issue in issues:
            target = _target_range(doc, paras, issue)
            if target is None:
                log.warning("未能标注(定位失败)：%s @ %s",
                            issue.rule_id, issue.location)
                continue
            try:
                target.CharBackColor = _RED
                _add_comment(doc, target, issue.message)
            except Exception:  # noqa: BLE001 - 单个 issue 失败不影响其余
                log.warning("未能标注(写入失败)：%s", issue.rule_id)
        url = uno.systemPathToFileUrl(output_docx)
        doc.storeToURL(url, (
            PropertyValue(Name="FilterName", Value="MS Word 2007 XML"),))
```

```python
"""串联三步：extract → rules → annotate。"""
from __future__ import annotations

import logging

from thesis_review.models import Issue
from thesis_review.extractor import extract
from thesis_review.annotator import annotate
from thesis_review.rules.registry import get_rules
from thesis_review.uno_session import UnoSession

log = logging.getLogger(__name__)


def review(input_docx: str, output_docx: str) -> list[Issue]:
    with UnoSession() as session:
        with session.open(input_docx) as doc:
            document = extract(doc, source_path=input_docx)
        issues: list[Issue] = []
        for rule in get_rules():
            try:
                issues.extend(rule.check(document))
            except Exception:  # noqa: BLE001 - 单条规则异常不影响其余
                log.exception("规则执行失败：%s", rule.rule_id)
        annotate(session, input_docx, output_docx, issues)
    return issues
```

```python
"""命令行入口。"""
from __future__ import annotations

import argparse
import sys
from pathlib import Path


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="thesis-review",
                                description="毕业设计论文格式审核")
    p.add_argument("input", help="输入 .docx 路径")
    p.add_argument("-o", "--output", required=True, help="输出 marked.docx 路径")
    return p


def validate_input(path: str) -> None:
    f = Path(path)
    if not f.exists():
        print(f"错误：文件不存在：{path}", file=sys.stderr)
        raise SystemExit(2)
    if f.suffix.lower() != ".docx":
        print("错误：仅支持 .docx 输入(不支持 .doc，不做转换)", file=sys.stderr)
        raise SystemExit(2)


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    validate_input(args.input)
    # 延迟导入：避免无 UNO 环境时 import 失败影响 --help
    from thesis_review.pipeline import review
    issues = review(args.input, args.output)
    errors = sum(1 for i in issues if i.severity == "error")
    warnings = sum(1 for i in issues if i.severity == "warning")
    print(f"检查完成：{errors} 处错误，{warnings} 处警告 → {args.output}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
