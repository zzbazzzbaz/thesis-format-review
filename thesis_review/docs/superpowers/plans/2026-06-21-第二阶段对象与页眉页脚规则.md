# 第二阶段对象与页眉页脚规则 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现第二阶段对象与页眉页脚规则：`global` 页眉/页脚页码格式，以及 `main-text` 公式、图片图题、表格表题规则。

**Architecture:** 保持 `UNO extractor -> DocumentModel -> RuleContext -> Checker -> IssueFactory` 链路。先扩展核心模型和测试构造器，再补齐 UNO 页眉页脚、页样式、图片、公式证据抽取；规则层只消费 `DocumentModel`，基于 section、块相邻关系、页样式和 run 格式判断，不读取样本名或 `case.yml`。

**Tech Stack:** Python 3.12, uv, pytest, Pydantic, PyYAML, LibreOffice/UNO, Docker.

"""引擎无关的核心数据模型，是各层之间唯一的耦合点。"""
from __future__ import annotations

from typing import Annotated, Literal

from pydantic import BaseModel, Field, model_validator

BlockKind = Literal["paragraph", "table", "image"]
TargetKind = Literal["block", "run"]
IssueCategory = Literal["format", "structure", "ai"]
IssueSeverity = Literal["error", "warning", "info"]
TabStopAlignment = Literal["left", "center", "right", "decimal", "unknown"]
PageNumberFormat = Literal["arabic", "roman", "unknown"]


class RunSpan(BaseModel):
    id: str
    text: str
    run_index: int
    font_east_asian: str | None
    font_latin: str | None
    size_pt: float | None
    bold: bool
    italic: bool
    superscript: bool = False


class FormattedTextRun(BaseModel):
    text: str
    run_index: int
    font_east_asian: str | None = None
    font_latin: str | None = None
    size_pt: float | None = None
    bold: bool = False
    italic: bool = False


class HeaderFooterParagraph(BaseModel):
    text: str
    paragraph_index: int
    alignment: str | None = None
    runs: list[FormattedTextRun] = []
    has_page_number_field: bool = False
    page_number_text: str | None = None
    page_number_format: PageNumberFormat | None = None


class TabStopInfo(BaseModel):
    position_mm: float | None = None
    alignment: TabStopAlignment = "unknown"


class BlockBase(BaseModel):
    id: str
    kind: BlockKind
    flow_index: int
    type_index: int
    page_style_name: str | None = None


class ParagraphBlock(BlockBase):
    kind: Literal["paragraph"] = "paragraph"
    text: str
    style_name: str | None
    outline_level: int | None
    alignment: str | None
    line_spacing: float | None
    first_line_indent_chars: float | None
    page_break_before: bool = False
    is_empty: bool = False
    runs: list[RunSpan]
    contains_math: bool = False
    math_object_count: int = 0
    tab_stops: list[TabStopInfo] = []


class TableBlock(BlockBase):
    kind: Literal["table"] = "table"
    row_count: int | None = None
    column_count: int | None = None
    text: str = ""

    kind: Literal["image"] = "image"
    name: str | None = None
    alt_text: str | None = None
    width_mm: float | None = None
    height_mm: float | None = None

    ParagraphBlock | TableBlock | ImageBlock,
    Field(discriminator="kind"),

def make_para(
    flow_index=0,
    type_index=None,
    text="正文",
    style_name="正文",
    outline_level=None,
    alignment="left",
    line_spacing=1.5,
    first_line_indent_chars=2.0,
    runs=None,
    is_empty=None,
    page_break_before=False,
    page_style_name=None,
    contains_math=False,
    math_object_count=0,
    tab_stops=None,
):
    if type_index is None:
        type_index = flow_index
    block_id = f"b-{flow_index:04d}"
    if is_empty is None:
        is_empty = not text.strip()
    if runs is None:
        runs = [] if is_empty else [make_run(text=text, block_id=block_id)]
    else:
        runs = [
            run.model_copy(update={"id": f"{block_id}-r-{idx:04d}", "run_index": idx})
            for idx, run in enumerate(runs)
        ]
    return ParagraphBlock(
        id=block_id,
        flow_index=flow_index,
        type_index=type_index,
        page_style_name=page_style_name,
        text=text,
        style_name=style_name,
        outline_level=outline_level,
        alignment=alignment,
        line_spacing=line_spacing,
        first_line_indent_chars=first_line_indent_chars,
        page_break_before=page_break_before,
        is_empty=is_empty,
        runs=runs,
        contains_math=contains_math,
        math_object_count=math_object_count,
        tab_stops=tab_stops or [],
    )

    name="正文",
    header_paragraphs=None,
    footer_paragraphs=None,

def section_blocks(context: RuleContext, name: str) -> list[DocumentBlock]:
    section = context.sections.get(name)
    if section is None:
        return []
    return list(blocks_in_range(context.document, section))


def display_page_style_actual(name: str | None) -> str:
    return name or "未识别"


def run_ranges(para: ParagraphBlock) -> list[tuple[int, int, RunSpan]]:
    ranges: list[tuple[int, int, RunSpan]] = []
    cursor = 0
    for run in para.runs:
        start = cursor
        cursor += len(run.text)
        ranges.append((start, cursor, run))
    return ranges


def runs_overlapping(para: ParagraphBlock, start: int, end: int) -> list[RunSpan]:
    return [
        run for run_start, run_end, run in run_ranges(para)
        if run_end > start and run_start < end and run.text.strip()
    ]
```

_ARABIC_RE = re.compile(r"^\d+$")
_ROMAN_RE = re.compile(r"^[IVXLCDM]+$", re.IGNORECASE)


def _styles_by_name(context: RuleContext) -> dict[str, PageStyleInfo]:
    return {style.name: style for style in context.document.page_styles}


def _first_blocks_by_style(
    context: RuleContext,
    section_names: tuple[str, ...],
) -> dict[str, DocumentBlock]:
    out: dict[str, DocumentBlock] = {}
    for section_name in section_names:
        for block in section_blocks(context, section_name):
            if block.page_style_name and block.page_style_name not in out:
                out[block.page_style_name] = block
    return out


def _visible_runs(paragraph: HeaderFooterParagraph) -> list[FormattedTextRun]:
    return [run for run in paragraph.runs if run.text.strip()]


def _paragraph_text(paragraphs: list[HeaderFooterParagraph]) -> str:
    return "\n".join(para.text for para in paragraphs if para.text.strip()).strip()


def _first_visible_paragraph(
    paragraphs: list[HeaderFooterParagraph],
) -> HeaderFooterParagraph | None:
    return next((para for para in paragraphs if para.text.strip()), None)


def _is_arabic_page_number(paragraph: HeaderFooterParagraph) -> bool:
    if paragraph.page_number_format == "arabic":
        return True
    text = (paragraph.page_number_text or paragraph.text).strip()
    return bool(_ARABIC_RE.fullmatch(text))


def _is_page_number_candidate(paragraph: HeaderFooterParagraph) -> bool:
    if paragraph.has_page_number_field:
        return True
    text = (paragraph.page_number_text or paragraph.text).strip()
    return bool(_ARABIC_RE.fullmatch(text) or _ROMAN_RE.fullmatch(text))


class GlobalHeaderChecker:
    checker_id = "global-header"
    rule_ids = (
        "global-header-001",
        "global-header-002",
        "global-header-003",
        "global-header-004",
    )

    def check(self, context: RuleContext) -> list[Issue]:
        issues: list[Issue] = []
        styles = _styles_by_name(context)
        for style_name, anchor in _first_blocks_by_style(context, HEADER_SECTIONS).items():
            style = styles.get(style_name)
            paragraphs = style.header_paragraphs if style is not None else []
            header_text = _paragraph_text(paragraphs)
            if not header_text:
                append_issue(
                    issues,
                    context,
                    "global-header-001",
                    "未识别",
                    block_location(anchor),
                )
                continue
            if header_text != EXPECTED_HEADER:
                append_issue(
                    issues,
                    context,
                    "global-header-001",
                    header_text,
                    block_location(anchor),
                )
            paragraph = _first_visible_paragraph(paragraphs)
            if paragraph is None:
                continue
            for run in _visible_runs(paragraph):
                if contains_chinese(run.text) and run.font_east_asian != "黑体":
                    append_issue(
                        issues,
                        context,
                        "global-header-002",
                        run.font_east_asian or "未识别",
                        block_location(anchor),
                    )
                    break
            for run in _visible_runs(paragraph):
                if run.size_pt != 10.5:
                    append_issue(
                        issues,
                        context,
                        "global-header-003",
                        display_size(run.size_pt),
                        block_location(anchor),
                    )
                    break
            if paragraph.alignment != "center":
                append_issue(
                    issues,
                    context,
                    "global-header-004",
                    display_alignment(paragraph.alignment),
                    block_location(anchor),
                )

def _chapter_number(para: ParagraphBlock) -> int | None:
    match = _H1_RE.match(para.text.strip())
    if match is None:
        return None
    numeral = match.group(2)
    if numeral.isdigit():
        return int(numeral)
    if numeral in _CHINESE_NUMERAL_MAP:
        return _CHINESE_NUMERAL_MAP[numeral]
    if numeral.startswith("十") and len(numeral) == 2:
        return 10 + _CHINESE_NUMERAL_MAP.get(numeral[1], 0)
    if numeral.endswith("十") and len(numeral) == 2:
        return _CHINESE_NUMERAL_MAP.get(numeral[0], 0) * 10
    return None

    current: int | None = None

def _is_formula_candidate(para: ParagraphBlock) -> bool:
    text = para.text.strip()
    if not text:
        return False
    if _is_heading(para) or _FIGURE_CAPTION_ANY_RE.match(text) or _TABLE_CAPTION_ANY_RE.match(text):
        return False
    if para.contains_math or para.math_object_count > 0:

    if para.style_name and para.style_name.lower() in {"公式", "equation"}:
        return True
    if "式" in text and len(text) > 20:
        return False
    return bool(("=" in text or "\t" in text) and _MATH_EXPR_RE.match(text))

        footer_paragraphs = _header_footer_paragraphs(getattr(style, "FooterText", None))
        styles.append(
            PageStyleInfo(
                name=name,
                width_mm=_mm100_to_mm(getattr(style, "Width", None)),
                height_mm=_mm100_to_mm(getattr(style, "Height", None)),
                margin_top_mm=_mm100_to_mm(getattr(style, "TopMargin", None)),
                margin_bottom_mm=_mm100_to_mm(getattr(style, "BottomMargin", None)),
                margin_left_mm=_mm100_to_mm(getattr(style, "LeftMargin", None)),
                margin_right_mm=_mm100_to_mm(getattr(style, "RightMargin", None)),
                header_text="\n".join(p.text for p in header_paragraphs if p.text.strip()) or None,
                footer_text="\n".join(p.text for p in footer_paragraphs if p.text.strip()) or None,
                header_paragraphs=header_paragraphs,
                footer_paragraphs=footer_paragraphs,
            )
        )
    return styles
```
