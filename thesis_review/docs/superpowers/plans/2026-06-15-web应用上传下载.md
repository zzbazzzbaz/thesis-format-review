# Web 应用上传下载 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有 `pipeline.review(input_docx, output_docx)` 外增加一个很薄的 FastAPI Web 层，让浏览器上传 `.docx` 后直接下载标注后的 `.marked.docx`。

**Architecture:** Web 层只做 HTTP、静态页面、临时文件和下载响应适配，不改 CLI 和审核 pipeline。`routes.py` 只处理请求/响应与错误码，`services.py` 集中处理上传文件落盘、临时目录生命周期、下载文件名和 `pipeline.review()` 调用，后续任务队列扩展只需要替换 service 边界。

**Tech Stack:** Python 3.12、FastAPI、Uvicorn、python-multipart、Starlette `FileResponse`/`BackgroundTask`、pytest、FastAPI `TestClient`/httpx、现有 LibreOffice/UNO Docker 测试镜像。

**执行目录:** 除特别说明外，所有命令从 `thesis_review/` 目录运行。

"""上传文件保存、审核调用和临时文件生命周期管理。"""
from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path
import shutil
import tempfile

from fastapi import UploadFile

from thesis_review import pipeline

CHUNK_SIZE = 1024 * 1024


class InvalidUpload(ValueError):
    """用户上传内容不符合首版 Web 接口要求。"""


@dataclass(frozen=True)
class ReviewResult:
    output_path: Path
    download_name: str
    cleanup: Callable[[], None]


def _client_filename(filename: str | None) -> str:
    if not filename:
        raise InvalidUpload("请选择 .docx 文件")
    name = filename.replace("\\", "/").split("/")[-1].strip()
    if not name:
        raise InvalidUpload("请选择 .docx 文件")
    if Path(name).suffix.lower() != ".docx":
        raise InvalidUpload("仅支持 .docx 文件")
    return name


def _download_name(filename: str) -> str:
    stem = Path(filename).stem
    return f"{stem}.marked.docx"


async def _save_upload(upload: UploadFile, target: Path) -> None:
    with target.open("wb") as f:
        while True:
            chunk = await upload.read(CHUNK_SIZE)
            if not chunk:
                break
            f.write(chunk)


async def review_upload(upload: UploadFile) -> ReviewResult:
    filename = _client_filename(upload.filename)
    workdir = Path(tempfile.mkdtemp(prefix="thesis-review-"))
    input_path = workdir / "input.docx"
    output_path = workdir / "output.marked.docx"

    def cleanup() -> None:
        shutil.rmtree(workdir, ignore_errors=True)

    try:
        await _save_upload(upload, input_path)
        pipeline.review(str(input_path), str(output_path))
    except Exception:
        cleanup()
        raise

    return ReviewResult(
        output_path=output_path,
        download_name=_download_name(filename),
        cleanup=cleanup,
    )
```

"""Web HTTP 路由。"""
from __future__ import annotations

from pathlib import Path

from fastapi import APIRouter, File, HTTPException, UploadFile
from starlette.background import BackgroundTask
from starlette.responses import FileResponse

from thesis_review.web import services

DOCX_MIME = "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
STATIC_DIR = Path(__file__).parent / "static"

router = APIRouter()


@router.get("/", include_in_schema=False)
def index() -> FileResponse:
    return FileResponse(STATIC_DIR / "index.html")


@router.post("/api/review")
async def review_endpoint(file: UploadFile = File(...)) -> FileResponse:
    try:
        result = await services.review_upload(file)
    except services.InvalidUpload as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    except Exception as exc:
        raise HTTPException(status_code=500, detail="审核失败，请稍后重试") from exc

    return FileResponse(
        result.output_path,
        media_type=DOCX_MIME,
        filename=result.download_name,
        background=BackgroundTask(result.cleanup),
    )
```

<!doctype html>
<html lang="zh-CN">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>毕业设计格式审核系统</title>
    <link rel="stylesheet" href="/static/style.css">
  </head>
  <body>
    <main class="shell">
      <form id="review-form">
        <div class="field">
          <label for="file">选择 Word 文档</label>
          <input id="file" name="file" type="file" accept=".docx" required>
        </div>
        <button id="submit" type="submit">开始审核</button>
        <p id="status" class="status" role="status" aria-live="polite"></p>
      </form>
    </main>
    <script>
      const form = document.querySelector("#review-form");
      const fileInput = document.querySelector("#file");
      const submit = document.querySelector("#submit");
      const status = document.querySelector("#status");

      function setStatus(message, className) {
        status.textContent = message;
        status.className = className ? `status ${className}` : "status";
      }

      function downloadBlob(blob, fallbackName, response) {
        const disposition = response.headers.get("content-disposition") || "";
        const match = disposition.match(/filename="([^"]+)"/);
        const filename = match ? match[1] : fallbackName;
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.href = url;
        link.download = filename;
        document.body.appendChild(link);
        link.click();
        link.remove();
        URL.revokeObjectURL(url);
      }

      form.addEventListener("submit", async (event) => {
        event.preventDefault();
        const file = fileInput.files[0];
        if (!file) {
          setStatus("请选择 .docx 文件", "error");
          return;
        }

        const data = new FormData();
        data.append("file", file);
        submit.disabled = true;
        setStatus("正在审核", "");

        try {
          const response = await fetch("/api/review", {
            method: "POST",
            body: data,
          });

          if (!response.ok) {
            let message = "审核失败";
            try {
              const payload = await response.json();
              if (payload.detail) {
                message = payload.detail;
              }
            } catch (error) {
              message = "审核失败";
            }
            setStatus(message, "error");
            return;
          }

          const blob = await response.blob();
          const stem = file.name.replace(/\.docx$/i, "");
          downloadBlob(blob, `${stem}.marked.docx`, response);
          setStatus("审核完成，已开始下载", "success");
        } catch (error) {
          setStatus("审核失败，请稍后重试", "error");
        } finally {
          submit.disabled = false;
        }
      });
    </script>
  </body>
</html>

from thesis_review.web.app import app

DOCX_MIME = "application/vnd.openxmlformats-officedocument.wordprocessingml.document"


@pytest.mark.uno
def test_web_review_endpoint_runs_pipeline_and_returns_docx(sample_docx, tmp_path):
    client = TestClient(app)

    with open(sample_docx, "rb") as f:
        response = client.post(
            "/api/review",
            files={"file": ("sample.docx", f, DOCX_MIME)},
        )

    assert response.status_code == 200
    assert response.headers["content-type"] == DOCX_MIME
    assert 'filename="sample.marked.docx"' in response.headers["content-disposition"]
    out = tmp_path / "sample.marked.docx"
    out.write_bytes(response.content)
    assert zipfile.is_zipfile(out)

    from thesis_review.uno_session import UnoSession

    with UnoSession() as session:
        with session.open(str(out)) as doc:
            assert doc.getText().getString()
