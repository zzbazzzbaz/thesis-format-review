```python
"""规则检查器共享格式工具。"""
from __future__ import annotations

import re
from collections.abc import Callable, Iterable
from math import isclose

from thesis_review.core.models import Issue, Location, ParagraphBlock, RunSpan
from thesis_review.core.sections import paragraphs_in_range
from thesis_review.rules.context import RuleContext

_ASCII_TEXT_RE = re.compile(r"[A-Za-z0-9]")
_CHINESE_TEXT_RE = re.compile(r"[\u4e00-\u9fff，。；：、（）《》“”‘’]")


def block_location(para: ParagraphBlock) -> Location:
    return Location(
        target="block",
        block_id=para.id,
        block_kind="paragraph",
        flow_index=para.flow_index,
    )


def run_location(para: ParagraphBlock, run: RunSpan) -> Location:
    return Location(
        target="run",
        block_id=para.id,
        block_kind="paragraph",
        flow_index=para.flow_index,
        run_id=run.id,
        run_index=run.run_index,
    )


def section_paragraphs(context: RuleContext, name: str) -> list[ParagraphBlock]:
    section = context.sections.get(name)
    if section is None:
        return []
    return list(paragraphs_in_range(context.document, section))


def non_empty(paragraphs: Iterable[ParagraphBlock]) -> list[ParagraphBlock]:
    return [para for para in paragraphs if para.text.strip()]


def empty_count_after(paragraphs: list[ParagraphBlock], para: ParagraphBlock) -> int:
    try:
        index = paragraphs.index(para)
    except ValueError:
        return 0
    count = 0
    for candidate in paragraphs[index + 1:]:
        if candidate.text.strip():
            break
        count += 1
    return count


def display_size(size_pt: float | None) -> str:
    if size_pt is None:
        return "未识别"
    return f"{size_pt:g}pt"


def display_line_spacing(line_spacing: float | None) -> str:
    if line_spacing is None:
        return "未识别"
    return f"{line_spacing:g} 倍"


def display_indent(indent: float | None) -> str:
    if indent is None:
        return "未识别"
    return f"{indent:g} 字符"


def display_alignment(alignment: str | None) -> str:
    names = {
        "center": "居中",
        "left": "左对齐",
        "right": "右对齐",
        "justify": "两端对齐",
    }
    return names.get(alignment or "", alignment or "未识别")


def has_size(run: RunSpan, size: float) -> bool:
    return run.size_pt is not None and isclose(run.size_pt, size, abs_tol=0.01)


def has_line_spacing(para: ParagraphBlock, line_spacing: float) -> bool:
    return (
        para.line_spacing is not None
        and isclose(para.line_spacing, line_spacing, abs_tol=0.01)
    )


def has_indent(para: ParagraphBlock, indent: float) -> bool:
    return (
        para.first_line_indent_chars is not None
        and isclose(para.first_line_indent_chars, indent, abs_tol=0.05)
    )


def meaningful_runs(para: ParagraphBlock) -> list[RunSpan]:
    return [run for run in para.runs if run.text.strip()]


def first_bad_run(
    para: ParagraphBlock,
    predicate: Callable[[RunSpan], bool],
) -> RunSpan | None:
    for run in meaningful_runs(para):
        if not predicate(run):
            return run
    return None


def contains_chinese(text: str) -> bool:
    return bool(_CHINESE_TEXT_RE.search(text))


def contains_ascii(text: str) -> bool:
    return bool(_ASCII_TEXT_RE.search(text))


def append_issue(
    issues: list[Issue],
    context: RuleContext,
    rule_id: str,
    actual: str | None,
    location: Location,
) -> None:
    issues.append(
        context.issue_factory.create(
            rule_id=rule_id,
            actual=actual,
            location=location,
        )
    )


def append_first_bad_run(
    issues: list[Issue],
    context: RuleContext,
    para: ParagraphBlock,
    rule_id: str,
    predicate: Callable[[RunSpan], bool],
    actual: Callable[[RunSpan], str | None],
) -> None:
    bad = first_bad_run(para, predicate)
    if bad is None:
        return
    append_issue(
        issues,
        context,
        rule_id=rule_id,
        actual=actual(bad),
        location=run_location(para, bad),
    )
```

def _first_with_norm_prefix(
    paragraphs: list[ParagraphBlock],
    prefixes: tuple[str, ...],
) -> ParagraphBlock | None:
    for para in paragraphs:
        text = norm_text(para.text)
        if any(text.startswith(prefix) for prefix in prefixes):
            return para
    return None


def _all_with_norm_prefix(
    paragraphs: list[ParagraphBlock],
    prefixes: tuple[str, ...],
) -> list[ParagraphBlock]:
    return [
        para for para in paragraphs
        if any(norm_text(para.text).startswith(prefix) for prefix in prefixes)
    ]


def _date_line(paragraphs: list[ParagraphBlock]) -> ParagraphBlock | None:
    for para in paragraphs:
        text = para.text
        if "年" in text and "月" in text and "日" in text:

    return None

import re

_BRACKET_PREFIX_RE = re.compile(r"^(\[[^\]]+\])([ ]*)")
_VALID_NUMBER_RE = re.compile(r"^\[\d+\]")


def _title_and_entries(context: RuleContext) -> tuple[list[ParagraphBlock], ParagraphBlock | None, list[ParagraphBlock]]:
    paragraphs = section_paragraphs(context, "references")
    visible = non_empty(paragraphs)
    if not visible:
        return paragraphs, None, []
    title = visible[0]
    start = paragraphs.index(title) + 1 + empty_count_after(paragraphs, title)
    entries = [para for para in paragraphs[start:] if para.text.strip()]
    return paragraphs, title, entries


def _is_chinese_font(run: RunSpan) -> bool:
    return not contains_chinese(run.text) or run.font_east_asian == "宋体"


def _is_latin_font(run: RunSpan) -> bool:
    return not contains_ascii(run.text) or run.font_latin == "Times New Roman"


class ReferencesTitleChecker:
    checker_id = "references-title"
    rule_ids = ("references-title-001", "references-title-002")

    def check(self, context: RuleContext) -> list[Issue]:
        paragraphs, title, _ = _title_and_entries(context)
        if title is None:
            return []
        issues: list[Issue] = []
        if title.text != "参考文献":
            append_issue(issues, context, "references-title-001", title.text, block_location(title))
        blank_count = empty_count_after(paragraphs, title)
        if blank_count != 1:
            append_issue(issues, context, "references-title-002", f"{blank_count} 个空段落", block_location(title))
        return issues


class ReferencesEntryChecker:
    checker_id = "references-entry"
    rule_ids = (
        "references-entry-001",
        "references-entry-002",
        "references-entry-003",
        "references-entry-004",
        "references-entry-005",
        "references-entry-006",
    )

    def check(self, context: RuleContext) -> list[Issue]:
        _, _, entries = _title_and_entries(context)
        issues: list[Issue] = []
        for para in entries:
            self._check_numbering(context, para, issues)
            self._check_format(context, para, issues)
        return issues

    def _check_numbering(self, context: RuleContext, para: ParagraphBlock, issues: list[Issue]) -> None:
        text = para.text
        if _VALID_NUMBER_RE.match(text) is None:
            actual = text.split(" ", 1)[0] if text.strip() else "未识别"
            append_issue(issues, context, "references-entry-001", actual, block_location(para))
        match = _BRACKET_PREFIX_RE.match(text)
        if match is None:
            append_issue(issues, context, "references-entry-002", "未找到编号后空格", block_location(para))
            return
        spaces = match.group(2)
        if len(spaces) != 1:
            append_issue(issues, context, "references-entry-002", f"{len(spaces)} 个空格", block_location(para))

    def _check_format(self, context: RuleContext, para: ParagraphBlock, issues: list[Issue]) -> None:
        append_first_bad_run(issues, context, para, "references-entry-003", _is_chinese_font, lambda run: run.font_east_asian or "未识别")

            append_issue(issues, context, "references-entry-006", display_line_spacing(para.line_spacing), block_location(para))

        return not contains_chinese(run.text) or run.font_east_asian == expected
    return predicate


def _is_latin_font(expected: str):
    def predicate(run: RunSpan) -> bool:
        return not contains_ascii(run.text) or run.font_latin == expected
    return predicate

    return para.text.startswith("第") and "章" in para.text and not is_toc_entry(para.text)


def _is_h2_candidate(para: ParagraphBlock) -> bool:
    text = para.text.strip()
    return bool(_H2_RE.match(text) or _MALFORMED_H2_RE.match(text))


def _is_h3_candidate(para: ParagraphBlock) -> bool:
    text = para.text.strip()
    return bool(_H3_RE.match(text) or _MALFORMED_H3_RE.match(text))


def _is_caption_or_formula(para: ParagraphBlock) -> bool:
    text = para.text.strip()
    return bool(
        _FIGURE_CAPTION_RE.match(text)
        or _TABLE_CAPTION_RE.match(text)
        or _FORMULA_LIKE_RE.match(text)
    )

def _run_ranges(para: ParagraphBlock) -> list[tuple[int, int, RunSpan]]:
    ranges: list[tuple[int, int, RunSpan]] = []
    cursor = 0
    for run in para.runs:
        start = cursor
        cursor += len(run.text)
        ranges.append((start, cursor, run))
    return ranges


def _runs_overlapping(para: ParagraphBlock, start: int, end: int) -> list[RunSpan]:
    return [

        if run_end > start and run_start < end and run.text.strip()
    ]
